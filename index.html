Explorando as Estrelas

Mova o mouse ou veja a magia acontecer.

const canvas = document.getElementById("space"); const ctx = canvas.getContext("2d"); canvas.width = window.innerWidth; canvas.height = window.innerHeight; const stars = []; const numStars = 200; for (let i = 0; i < numStars; i++) { stars.push({ x: Math.random() * canvas.width, y: Math.random() * canvas.height, radius: Math.random() * 1.5, alpha: Math.random(), speed: Math.random() * 0.02 }); } function animate() { ctx.clearRect(0, 0, canvas.width, canvas.height); stars.forEach(star => { star.alpha += star.speed; if (star.alpha > 1 || star.alpha < 0) { star.speed = -star.speed; } ctx.beginPath(); ctx.arc(star.x, star.y, star.radius, 0, Math.PI * 2); ctx.fillStyle = `rgba(255, 255, 255, ${Math.abs(star.alpha)})`; ctx.fill(); }); requestAnimationFrame(animate); } animate(); def desenhar_estrela(linhas): print("\n--- Pirâmide de Estrelas ---\n") for i in range(1, linhas + 1): # Imprime os espaços e depois as estrelas espacos = " " * (linhas - i) estrelas = "*" * (2 * i - 1) print(espacos + estrelas) # Exemplo de uso desenhar_estrela(5)
